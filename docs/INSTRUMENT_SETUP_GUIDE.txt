%% INSTRUMENT SETUP GUIDE
% Canonical setup scripts:
% - Recipe (worker engine): `demos/demo.m`
% - Recipe (worker engine, n/E example, no hardware required): `demos/demo_nE.m`
% - Legacy rack-on-client debug example (git-ignored): `tests/demo_nE.m`
%
% Keep the relevant demo(s) up to date when you add/change instruments/channels.
%
% QUICK REFERENCE
% ---------------
% 0) Add sm-dev or sm-main code folder to the MATLAB path
% 1) Run sminit (shared setup: globals, cleanup, UI state)
% 2) Create instrument object (handle)
% 3) Optional: set instrument properties / send direct hardware commands
% 4) Add instrument to rack
% 5) Add user-facing channels to rack
% 6) smready(rack)  OR  smready(recipe)
%
% STEP 0: Add sm-dev or sm-main code folder to the path
% This matches tests/demo_nE.m and keeps sminit available.
% Example:
%   username = string(getenv("USERNAME"));
%   sm_dev_path = fullfile("C:\Users", username, "Desktop", "sm-dev");
%   if exist(sm_dev_path, "dir")
%       codePath = fullfile(sm_dev_path, "code");
%       addpath(codePath);
%   end
%
% STEP 1: Run shared initialization
%   sminit;
%
% STEP 2: Create instrument object
% Syntax: handle = instrument_ClassName(address)
%
% Examples:
%   handle_clock = instrument_clock("clock");
%   handle_K2450 = instrument_K2450(gpibAddress(18, 0));
%   handle_SR830 = instrument_SR830(gpibAddress(7, 0));
%   handle_Montana2 = instrument_Montana2("192.168.1.100");
%
% STEP 3 (optional): configure instrument
% Common patterns:
%   handle.requireSetCheck = false;  % skip set verification for fast/unsafe cases
%   h = handle.communicationHandle;
%   writeline(h, "SCPI_COMMAND");
%   resp = writeread(h, "SCPI_QUERY?");
% Notes:
% - This can be done before or after rack.addInstrument(...) (demo.m does both).
% - Prefer instrument class methods when available; use direct commands for gaps.
% - Safety default: keep `reset()` calls commented out unless you have
%   explicitly ramped outputs to a known-safe state.
% - Framework write pacing is controlled by `handle.writeCommandInterval`
%   (default `seconds(0)`). No automatic GPIB delay is applied by default;
%   set this manually per instrument only when needed.
% - Keithley timing note: consecutive gets/sets can be fast, but the read
%   phase after a get query can be much slower (often around 0.5 s,
%   depending on NPLC/averaging/auto-zero/readback settings).
%
% STEP 4: add instrument to rack
% Syntax: rack.addInstrument(handle, "instName")
%
% Examples:
%   rack.addInstrument(handle_clock, "clock");
%   rack.addInstrument(handle_K2450, "K2450_A");
%
% STEP 5: add channels to rack (user-facing names used by smget/smset)
% Full signature (matches explicit-rack debug scripts like tests/demo_nE.m):
%   rack.addChannel("instName", "instChanName", "userChanName", rampRate, rampThreshold, softMin, softMax)
%
% Notes:
% - rampRate, rampThreshold, softMin, softMax are optional.
% - Use [] to skip an optional argument position.
% - softMin/softMax are enforced by instrumentRack (smset/rackSet) only.
%   Direct instrument.setChannel calls bypass these software limits.
%
% Examples:
%   rack.addChannel("clock", "timeStamp", "time");
%   rack.addChannel("K2450_A", "V_source", "V_bg", 1, 0.5, -10, 10);
%   rack.addChannel("SR830_1", "XTheta", "Ixx_XTheta");
%
% STEP 6: finish setup + launch GUI
%   smready(rack);
%
% RECIPE MODE (worker engine)
% ---------------------------
% For recipe-based setups (recommended), build an `instrumentRackRecipe` instead:
%   recipe = instrumentRackRecipe();
%   recipe.addInstrument("h", "instrument_ClassName", "friendlyName", ...);
%   recipe.addChannel("friendlyName", "instChanName", "userChanName");
%   recipe.addStatement("friendlyName", "h = handle_myInst.communicationHandle;");
%   recipe.addStatement("friendlyName", "writeline(h, ""SCPI_COMMAND"");");
%   smready(recipe);
%
% Statement style guideline:
% - Prefer one executable line per `recipe.addStatement(...)` call.
% - This makes it easy to comment out individual setup lines in demos/scripts.
% - Statements run immediately after the referenced instrument is set up.
%
% Optional local-debug path with the same recipe (no engine worker):
%   smready(recipe, singleThreaded = true);
%
% RACK SCRIPT -> RECIPE SCRIPT MIGRATION
% --------------------------------------
% Use this when converting a legacy main script that currently does:
%   rack = instrumentRack(...); ... smready(rack);
%
% 1) Replace rack construction:
%      rack = instrumentRack(true);
%   -> recipe = instrumentRackRecipe();
%
% 2) Convert each physical-instrument block:
%      handle_K2450 = instrument_K2450(gpibAddress(18, 0));
%      handle_K2450.requireSetCheck = false;
%      h = handle_K2450.communicationHandle;
%      writeline(h, ":OUTPut ON");
%      rack.addInstrument(handle_K2450, "K2450_A");
%   -> recipe.addInstrument("handle_K2450", "instrument_K2450", "K2450_A", gpibAddress(18, 0));
%      recipe.addStatement("K2450_A", "handle_K2450.requireSetCheck = false;");
%      recipe.addStatement("K2450_A", "h = handle_K2450.communicationHandle;");
%      recipe.addStatement("K2450_A", "writeline(h, ':OUTPut ON');");
%
% 3) Convert channels 1:1:
%      rack.addChannel(...) -> recipe.addChannel(...)
%
% 4) Convert virtual-instrument construction:
%      handle_v = virtualInstrument_nE("virtual_nE", instrumentRackProxy(rack), ...);
%      rack.addInstrument(handle_v, "virtual_nE");
%   -> recipe.addVirtualInstrument("handle_v", "virtualInstrument_nE", "virtual_nE", "virtual_nE", ...);
%
% 5) Replace final setup call:
%      smready(rack);
%   -> smready(recipe);                         % worker engine default
%      smready(recipe, singleThreaded = true); % local debug with same recipe
%
% 6) If an instrument needs extra client workers, make it explicit in recipe:
%      recipe.addInstrument(..., numWorkersRequested = N);
%    In the instrument class, set `obj.numWorkersRequired = N` and use
%    `instrumentWorker(...)` for the actual worker startup/handshake.
%
% K33512B PAIRED OUTPUTS
% ----------------------------
% Enable K33512B_Use in your copied demos/demo.m and set its GPIB
% address/adaptorIndex_FG. The demo sets requireSetCheck = false; standalone
% instances inherit the instrumentInterface default (true).
% Set it true to verify BOTH channels after each set: frequency and amplitude
% must match the requested value on both; DC must match +value/-value.
% With it false, even an explicit setCheckChannel call skips readback and returns
% true, following instrumentInterface behavior. Enable it before a manual check.
% frequency / FG_Freq: Hz, writes the same frequency to CH1 and CH2.
% amplitude / FG_V_exc: Vrms per channel, before external dividers; writes both.
% dc_offset / FG_V_dc: volts, writes +value to CH1 and -value to CH2.
% Startup preserves CH1's existing DC offset and commands CH2 to its negative,
% without a forced readback check. Each ramp step also writes that +/- pair.
% The demo uses rampRate = 0.5 V/s and rampThreshold = 0.5 V for FG_V_dc.
% The demo sets both output loads to High Z (OUTP1:LOAD INF / OUTP2:LOAD INF)
% and both amplitude units to VRMS. It then reapplies the opposite DC pair using
% CH1's rescaled offset, since load changes rescale voltage values.
% For sine waves into high impedance on both channels, demo soft limits are:
%   FG_V_exc: 2e-3/(2*sqrt(2)) to 20/(2*sqrt(2)) Vrms per channel
%             (about 0.707 mVrms to 7.071 Vrms; 2 mVpp to 20 Vpp).
%   FG_V_dc: -10 to +10 V.
% These independent limits do not enforce the combined voltage envelope:
% abs(V_dc) + sqrt(2)*V_exc must also fit within 10 V per channel. The two
% independent maxima cannot be used together. Hardware range/error checks
% remain active; use a lower AC amplitude when scanning large DC offsets.
% Different waveforms or output-load settings require different soft limits.
% Reads report CH1 settings, not measured output voltages or a differential value.
% Startup sends OUTP2:POL INV to invert CH2's AC waveform around its DC offset.
% Polarity inversion itself leaves DC unchanged; startup and dc_offset separately
% set opposite DC offsets. Configure waveform, output enable and phase manually;
% set CH1 polarity Normal for opposite AC outputs with matching, aligned waveforms.
% Tracking and Ampl Cpl must be OFF. Freq Cpl may be OFF, or ON with ratio 1 /
% offset 0 Hz. Equal frequencies alone do not establish relative phase.
% An internal mirror mode exists (TRAC ON / INV), but TRAC INV copies the DC
% offset unchanged while inverting AC. Separate writes are needed for opposite DC.
% Amplitude access selects VRMS units on the instrument (also on the front panel).
% Arbitrary waveform sequences cannot use VRMS and produce an instrument error.
% The configured output load must match the real load for correct voltage scaling.
% Use smget/smset for scans and software limits. For standalone bring-up, use the
% inherited getChannel/setChannel methods; these bypass rack software limits.
% Set checks use setTolerances (default 1e-6 Hz or V) as an absolute floor,
% configurable with setSetTolerances. Effective tolerances also cover resolution:
%   frequency: max(setTolerances, 1e-6 Hz), from the 1-uHz specification;
%   amplitude / dc_offset: max(setTolerances, 0.001*abs(requested value)).
% The voltage allowance conservatively covers rounding or truncation to the
% specified four digits, including Vpp-to-Vrms conversion. It scales with the
% setpoint instead of applying a full-range millivolt tolerance to small signals.
% A small floating-point margin (8 eps at the target scale) is added to both.
% The manual does not specify the exact SCPI quantization algorithm; this is a
% resolution allowance, not the analog accuracy specification. Both readbacks
% must match their own target (+/- for DC); a mismatch returns false. The driver
% keeps this small override because the inherited check accepts only a fixed
% absolute tolerance and checks CH1 only. Readbacks do not measure AC settling.
% Both hardware ranges are checked before either setpoint is sent. Only numeric
% reply roundoff (64 eps at each boundary) is allowed past a queried boundary;
% the transmitted value is then clamped to that boundary. The larger set-check
% tolerance is NOT used to admit
% out-of-range requests. Paired writes are sequential, not atomic.
% Amplitude/offset conflicts can adjust the other parameter; an instrument error
% or partial write requires inspecting both outputs. Do not assume automatic
% rollback. Errors are reported without resets.
% Opposite DC is commanded at startup and at every DC ramp step;
% it cannot be guaranteed during the brief sequential update or after manual
% front-panel changes. Enable requireSetCheck=true to verify completed settings.
% Programming reference: Keysight Trueform Series Operating and Service Guide,
% OUTPut:LOAD, OUTPut:POLarity, FREQuency:COUPle, TRACk, VOLTage, VOLTage:OFFSet and
% VOLTage:UNIT sections:
% https://www.keysight.com/us/en/assets/9018-03714/service-manuals/9018-03714.pdf
% Resolution reference: 33500B/33600A data sheet, Waveform characteristics / Sine
% and Output characteristics / Amplitude / DC offset:
% https://www.keysight.com/us/en/assets/7018-05928/data-sheets/5992-2572.pdf
%
% STRAIN CELL CALIBRATION
% -----------------------
% instrument_strainController accepts strainCellNumber = 1, 2, or 3.
% Cell 3 is Razorbill CS130 #0509: alpha = 59.362 um pF, Cp = 0.01782 pF.
% demos/demo.m defaults to cell 1 for OptiCool and cell 3 for Montana2.
% Set strainCellNumber_default to match the installed cell if it changes.
% The worker uses alpha/(C-Cp) and subtracts the measured tare for del_d.
% The manufacturer's fixed d_0 = 62.257 um cancels in that difference;
% it is distinct from the controller's measured tare parameter d_0.
% When changing cells in an existing experiment folder, stop scans and keep
% activeControl off, then replace the old tare using the demo's handle:
%   global engine
%   engine.evalOnEngine("handle_strainController.tare()");
%
% COMMON PITFALLS
% --------------
% - addInstrument arg order is (handle, "instName"), not ("instName", handle).
% - The "instName" string must match between addInstrument and addChannel.
% - "instChanName" must exist in the instrument class (see its constructor/addChannel calls).
% - All names must be unique (instrument names, userChanName).
% - In demo scripts, keep the same instrument order across address variables,
%   `_Use` flags, and `if ..._Use` instrument blocks to keep maintenance predictable.
